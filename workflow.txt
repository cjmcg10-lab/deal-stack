name: Deal Stack collector
on:
  schedule:
    - cron: '17 */6 * * *'
  workflow_dispatch:
permissions:
  contents: write
jobs:
  collect:
    runs-on: ubuntu-latest
    timeout-minutes: 40
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-node@v5
        with:
          node-version: 22
      - name: Collect deals
        env:
          HUB_URL: ${{ secrets.HUB_URL }}
          HUB_KEY: ${{ secrets.HUB_KEY }}
        run: node collector.js
      - name: Keep the schedule switched on
        run: |
          last=$(git log -1 --format=%ct)
          now=$(date +%s)
          if [ $(( (now - last) / 86400 )) -ge 45 ]; then
            date -u > keepalive.txt
            git config user.name "deal-stack-bot"
            git config user.email "deal-stack-bot@users.noreply.github.com"
            git add keepalive.txt
            git commit -m "Keep the scheduled collector active"
            git push
          fi
